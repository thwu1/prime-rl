# Syzkaller syscall descriptions for /dev/dmabuf_mgr
# Describes the DMA buffer management device interface.

include <linux/types.h>

# Resource definitions
resource fd_dmabuf[fd]
resource dmabuf_handle[int64]: 0xffffffffffffffff

# Type aliases
type dmabuf_alignment int16[0:12]

# Flag definitions
dmabuf_flags = DMABUF_FLAG_READ, DMABUF_FLAG_WRITE, DMABUF_FLAG_EXEC, DMABUF_FLAG_COHERENT, DMABUF_FLAG_CACHED
dmabuf_mem_types = DMABUF_MEM_TYPE_SYSTEM, DMABUF_MEM_TYPE_DEVICE, DMABUF_MEM_TYPE_COHERENT
dmabuf_sync_flags = DMABUF_FLAG_READ, DMABUF_FLAG_WRITE, DMABUF_SYNC_START, DMABUF_SYNC_END
dmabuf_xfer_dir = DMABUF_XFER_TO_DEVICE, DMABUF_XFER_FROM_DEVICE, DMABUF_XFER_BIDIRECTIONAL

# Open device
openat$dmabuf_mgr(fd const[AT_FDCWD], file ptr[in, string["/dev/dmabuf_mgr"]], flags flags[open_flags], mode const[0]) fd

# Ioctl definitions
ioctl$DMABUF_IOC_CREATE(fd fd_dmabuf, cmd const[DMABUF_IOC_CREATE], arg ptr[in, dmabuf_create_req])
ioctl$DMABUF_IOC_DESTROY(fd fd_dmabuf, cmd const[DMABUF_IOC_DESTROY], arg ptr[in, int32])
ioctl$DMABUF_IOC_MAP(fd fd_dmabuf, cmd const[DMABUF_IOC_MAP], arg ptr[inout, dmabuf_map_req])
ioctl$DMABUF_IOC_UNMAP(fd fd_dmabuf, cmd const[DMABUF_IOC_UNMAP], arg ptr[in, dmabuf_map_req])
ioctl$DMABUF_IOC_SYNC(fd fd_dmabuf, cmd const[DMABUF_IOC_SYNC], arg ptr[in, dmabuf_sync_req])
ioctl$DMABUF_IOC_TRANSFER(fd fd_dmabuf, cmd const[DMABUF_IOC_TRANSFER], arg ptr[in, dmabuf_transfer_req])
ioctl$DMABUF_IOC_QUERY(fd fd_dmabuf, cmd const[DMABUF_IOC_QUERY], arg ptr[inout, dmabuf_query_resp])
ioctl$DMABUF_IOC_BATCH(fd fd_dmabuf, cmd const[DMABUF_IOC_BATCH], arg ptr[in, dmabuf_batch_op])

close$dmabuf_mgr(fd fd_dmabuf)
